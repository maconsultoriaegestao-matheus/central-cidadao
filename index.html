// ══════════════════════════════════════════════════════════════ // COZINHA CENTRAL — Google Apps Script v2.5 // Planilha 02_Producao // Novidade v2.4: monitoramento de produções em andamento // ══════════════════════════════════════════════════════════════ function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if(data.acao === 'atualizarStatus') return atualizarStatus(ss, data); if(data.acao === 'excluirLinha') return excluirLinhas(ss, data); if(data.acao === 'salvarAndamento') return salvarAndamento(ss, data); if(data.acao === 'removerAndamento') return removerAndamento(ss, data); // Inserção genérica de linhas (Producoes, Separacoes, Desperdicios, etc.) const aba = ss.getSheetByName(data.aba); if(!aba) return resposta(false, 'Aba não encontrada: ' + data.aba); data.linhas.forEach(linha => aba.appendRow(linha)); return resposta(true, data.linhas.length + ' linha(s) inserida(s) em ' + data.aba); } catch(err) { return resposta(false, err.toString()); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const tipo = e.parameter.tipo; if(tipo === 'separacoes_abertas') return lerSeparacoesAbertas(ss); if(tipo === 'producoes_ativas') return lerProducoesAtivas(ss); if(tipo === 'producoes_andamento') return lerProducoesAndamento(ss); if(tipo === 'ultimo_seq') return lerUltimoSeq(ss); return resposta(true, 'Script ativo.'); } catch(err) { return resposta(false, err.toString()); } } // ══════════════════════════════════════════════ // MONITORAMENTO — produções em andamento // ══════════════════════════════════════════════ // GET ?tipo=producoes_andamento // Retorna todas as fichas ativas (em produção ou standby) function lerProducoesAndamento(ss) { const aba = garantirAbaAndamento(ss); if(aba.getLastRow() < 2) return respostaJSON([]); const dados = aba.getDataRange().getValues(); const headers = dados[0]; const lista = dados.slice(1) .filter(r => r.some(v => v !== '')) .map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); return { id: String(obj['ID'] || '').trim(), produto: String(obj['Produto'] || '').trim(), operador: String(obj['Operador'] || '').trim(), sepCodigo: String(obj['CodSep'] || '').trim(), status: String(obj['Status'] || '').trim(), inicio: String(obj['Inicio'] || '').trim(), duracaoSeg: parseInt(obj['DuracaoSeg'] || 0) }; }) .filter(r => r.id && r.produto); // filtra linhas vazias return respostaJSON(lista); } // POST acao=salvarAndamento // Faz upsert: atualiza se ID já existe, insere se não existe function salvarAndamento(ss, data) { const aba = garantirAbaAndamento(ss); const dados = aba.getDataRange().getValues(); const headers = dados[0]; const iId = headers.findIndex(h => h === 'ID'); // Procura linha existente com mesmo ID for(let i = 1; i < dados.length; i++) { if(String(dados[i][iId]).trim() === String(data.id).trim()) { // Atualiza linha existente const linha = [ data.id, data.produto || '', data.operador || '', data.sepCodigo || '', data.status || '', data.inicio || '', data.duracaoSeg || 0 ]; aba.getRange(i + 1, 1, 1, linha.length).setValues([linha]); return resposta(true, 'Andamento atualizado: ' + data.id); } } // Não encontrou — insere nova linha aba.appendRow([ data.id, data.produto || '', data.operador || '', data.sepCodigo || '', data.status || '', data.inicio || '', data.duracaoSeg || 0 ]); return resposta(true, 'Andamento inserido: ' + data.id); } // POST acao=removerAndamento // Remove a linha com o ID informado function removerAndamento(ss, data) { const aba = garantirAbaAndamento(ss); if(aba.getLastRow() < 2) return resposta(true, 'Nada a remover.'); const dados = aba.getDataRange().getValues(); const iId = dados[0].findIndex(h => h === 'ID'); for(let i = dados.length - 1; i >= 1; i--) { if(String(dados[i][iId]).trim() === String(data.id).trim()) { aba.deleteRow(i + 1); return resposta(true, 'Andamento removido: ' + data.id); } } return resposta(true, 'ID não encontrado (já removido?): ' + data.id); } // Garante que a aba Producoes_Andamento existe com os cabeçalhos corretos function garantirAbaAndamento(ss) { let aba = ss.getSheetByName('Producoes_Andamento'); if(!aba) { aba = ss.insertSheet('Producoes_Andamento'); aba.appendRow(['ID', 'Produto', 'Operador', 'CodSep', 'Status', 'Inicio', 'DuracaoSeg']); // Formata cabeçalho aba.getRange(1, 1, 1, 7).setFontWeight('bold').setBackground('#234760').setFontColor('#ffffff'); aba.setFrozenRows(1); } return aba; } // ══════════════════════════════════════════════ // SEPARAÇÕES // ══════════════════════════════════════════════ function lerSeparacoesAbertas(ss) { const aba = ss.getSheetByName('Separacoes'); if(!aba) return respostaJSON([]); const dados = aba.getDataRange().getValues(); if(dados.length < 2) return respostaJSON([]); const headers = dados[0]; const iCodigo = headers.findIndex(h => norm(h) === 'codigo'); const iData = headers.findIndex(h => norm(h) === 'data'); const iOp = headers.findIndex(h => norm(h) === 'operador'); const iStatus = headers.findIndex(h => norm(h) === 'status'); const separacoes = dados.slice(1) .filter(r => r.some(v => v !== '') && norm(r[iStatus]) === 'aberta' && String(r[iCodigo]||'').trim()) .map(r => ({ codigo: String(r[iCodigo]).trim(), data: formatarData(r[iData]), operador: String(r[iOp]||'').trim(), status: 'aberta', itens: [] })); if(!separacoes.length) return respostaJSON([]); // Busca itens da aba Separacoes_Itens // Agrupa por CodSep+Receita para evitar duplicatas (formato novo: 1 linha por insumo) const abaItens = ss.getSheetByName('Separacoes_Itens'); if(abaItens && abaItens.getLastRow() > 1) { const itensData = abaItens.getDataRange().getValues(); const hItens = itensData[0]; const iCodSep = hItens.findIndex(h => norm(h).includes('codigo') || norm(h).includes('separac')); const iReceita = hItens.findIndex(h => norm(h).includes('receita')); const iQtd = hItens.findIndex(h => norm(h) === 'qtd' || norm(h).includes('qtd plan') || (norm(h).includes('qtd') && !norm(h).includes('insumo'))); const iUnid = hItens.findIndex(h => (norm(h) === 'unid' || norm(h).includes('unidade')) && !norm(h).includes('insumo')); // Agrupa por CodSep+Receita — elimina duplicatas geradas pelo novo formato (1 linha/insumo) const mapaItens = {}; itensData.slice(1).filter(r => r.some(v => v !== '')).forEach(row => { const codSep = String(row[iCodSep] || '').trim(); const receita = String(row[iReceita] || '').trim(); const qtd = row[iQtd] || 0; const unid = String(row[iUnid] || '').trim(); if(!codSep || !receita) return; const chave = codSep + '||' + receita; if(!mapaItens[chave]) mapaItens[chave] = { codSep, nome: receita, qtd, unid }; }); // Empurra um item por receita (não por insumo) Object.values(mapaItens).forEach(item => { const sep = separacoes.find(s => s.codigo === item.codSep); if(sep) sep.itens.push({ nome: item.nome, qtd: item.qtd, unid: item.unid }); }); } return respostaJSON(separacoes); } function lerUltimoSeq(ss) { const aba = ss.getSheetByName('Separacoes'); if(!aba || aba.getLastRow() < 2) return respostaJSON({ultimo: 0}); const dados = aba.getDataRange().getValues(); const headers = dados[0]; const iCodigo = headers.findIndex(h => norm(h) === 'codigo'); let maximo = 0; dados.slice(1).filter(r => r.some(v => v !== '')).forEach(r => { const cod = String(r[iCodigo]||'').trim(); const match = cod.match(/SEP-\d{4}-(\d+)/); if(match) { const num = parseInt(match[1]); if(num > maximo) maximo = num; } }); return respostaJSON({ultimo: maximo}); } // ══════════════════════════════════════════════ // PRODUÇÕES // ══════════════════════════════════════════════ function lerProducoesAtivas(ss) { const aba = ss.getSheetByName('Producoes'); if(!aba) return respostaJSON([]); const dados = aba.getDataRange().getValues(); if(dados.length < 2) return respostaJSON([]); const headers = dados[0]; const iStatus = headers.findIndex(h => norm(h) === 'status'); if(iStatus < 0) return respostaJSON([]); const rows = dados.slice(1) .filter(r => r[iStatus] !== 'concluida' && r.some(v => v !== '')) .map(r => { const obj = {}; headers.forEach((h,i) => obj[h] = r[i]); return obj; }); return respostaJSON(rows); } // ══════════════════════════════════════════════ // UTILITÁRIOS // ══════════════════════════════════════════════ function excluirLinhas(ss, data) { const aba = ss.getSheetByName(data.aba); if(!aba) return resposta(false, 'Aba não encontrada: ' + data.aba); const dados = aba.getDataRange().getValues(); const headers = dados[0].map(h => norm(h)); const colCodigo = headers.findIndex(h => h === norm(data.colunaCodigo||'codigo')); if(colCodigo < 0) return resposta(false, 'Coluna não encontrada: ' + data.colunaCodigo); let deletadas = 0; for(let i = dados.length - 1; i >= 1; i--){ if(String(dados[i][colCodigo]).trim() === String(data.codigo).trim()){ aba.deleteRow(i + 1); deletadas++; } } return resposta(true, deletadas + ' linha(s) excluída(s) de ' + data.aba); } function atualizarStatus(ss, data) { const aba = ss.getSheetByName(data.aba); if(!aba) return resposta(false, 'Aba não encontrada: ' + data.aba); const dados = aba.getDataRange().getValues(); const headers = dados[0].map(h => norm(h)); const colCodigo = headers.findIndex(h => h === 'codigo'); const colStatus = headers.findIndex(h => h === 'status'); if(colCodigo < 0 || colStatus < 0) return resposta(false, 'Colunas não encontradas'); for(let i = 1; i < dados.length; i++) { if(String(dados[i][colCodigo]).trim() === String(data.codigo).trim()) { aba.getRange(i + 1, colStatus + 1).setValue(data.novoStatus); return resposta(true, 'Status atualizado: ' + data.codigo + ' → ' + data.novoStatus); } } return resposta(false, 'Código não encontrado: ' + data.codigo); } function norm(s) { return String(s||'').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'').trim(); } function formatarData(val) { if(!val) return ''; try { if(typeof val === 'string' && val.match(/^\d{2}\/\d{2}\/\d{4}$/)) return val; const d = new Date(val); if(isNaN(d.getTime())) return String(val); return String(d.getDate()).padStart(2,'0') + '/' + String(d.getMonth()+1).padStart(2,'0') + '/' + d.getFullYear(); } catch(e) { return String(val); } } function resposta(ok, msg) { return ContentService .createTextOutput(JSON.stringify({ ok, msg })) .setMimeType(ContentService.MimeType.JSON); } function respostaJSON(dados) { return ContentService .createTextOutput(JSON.stringify({ ok: true, dados })) .setMimeType(ContentService.MimeType.JSON); }
